let lang = 'ar';
let stage = 1;
let score = 0;
let timeLeft = 40;
let timerId = null;

function initGame() {
    stage = 1;
    score = 0;
    document.getElementById('scoreNum').innerText = score;
    nextQuestion();
}

function startTimer() {
    clearInterval(timerId);
    timeLeft = 40;
    document.getElementById('timer').innerText = `⏱️ ${timeLeft}`;
    timerId = setInterval(() => {
        timeLeft--;
        document.getElementById('timer').innerText = `⏱️ ${timeLeft}`;
        if (timeLeft <= 0) {
            clearInterval(timerId);
            alert(lang === 'ar' ? "انتهى الوقت! خسرت المحاولة." : "Time's up! Game over.");
            initGame();
        }
    }, 1000);
}

function nextQuestion() {
    if (stage > 4) {
        clearInterval(timerId);
        alert(lang === 'ar' ? `🎉 مبروك! ختمت اللعبة بنجاح! النقاط: ${score}` : `🎉 Congrats! You beat the game! Score: ${score}`);
        initGame();
        return;
    }

    document.getElementById('stageNum').innerText = stage;
    startTimer();

    const pool = questionsData[lang][`stage${stage}`];
    const q = pool[Math.floor(Math.random() * pool.length)];

    document.getElementById('qTitle').innerText = q.q;
    document.getElementById('qImage').src = q.img;

    const grid = document.getElementById('optionsGrid');
    grid.innerHTML = '';

    q.options.forEach((opt, idx) => {
        const btn = document.createElement('button');
        btn.innerText = opt;
        btn.onclick = () => {
            if (idx === q.correct) {
                score += stage * 25;
                document.getElementById('scoreNum').innerText = score;
                stage++;
                nextQuestion();
            } else {
                clearInterval(timerId);
                alert(lang === 'ar' ? "إجابة خاطئة! ستبدأ من جديد." : "Wrong answer! Restarting...");
                initGame();
            }
        };
        grid.appendChild(btn);
    });
}

function changeLanguage() {
    lang = document.getElementById('langSelect').value;
    document.dir = lang === 'ar' ? 'rtl' : 'ltr';
    document.getElementById('stageTxt').innerText = lang === 'ar' ? 'المرحلة:' : 'Stage:';
    document.getElementById('scoreTxt').innerText = lang === 'ar' ? 'النقاط:' : 'Score:';
    initGame();
}

function shareGame() {
    const msg = encodeURIComponent("العب معي الآن في لعبة تحديات مازن وتحدى معلوماتك! 🔥");
    window.open(`https://api.whatsapp.com/send?text=${msg}%20${window.location.href}`, '_blank');
}

function createMultiplayerRoom() {
    const peer = new Peer();
    peer.on('open', (id) => {
        const inviteUrl = `${window.location.href}?room=${id}`;
        navigator.clipboard.writeText(inviteUrl);
        alert(lang === 'ar' ? "تم نسخ رابط الدعوة! أرسله لصديقك للعب معك." : "Invite link copied to clipboard!");
    });
}
